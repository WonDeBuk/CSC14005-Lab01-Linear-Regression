% \preparedbox{<table macro>}{<row>}{<plot options>}: one boxplot from a row
% with columns lw, lq, med, uq, uw (see box_stats in analysis/plot_data.py).
\newcommand{\preparedbox}[3]{%
  \pgfplotstablegetelem{#2}{lw}\of#1\let\bxlw\pgfplotsretval
  \pgfplotstablegetelem{#2}{lq}\of#1\let\bxlq\pgfplotsretval
  \pgfplotstablegetelem{#2}{med}\of#1\let\bxmed\pgfplotsretval
  \pgfplotstablegetelem{#2}{uq}\of#1\let\bxuq\pgfplotsretval
  \pgfplotstablegetelem{#2}{uw}\of#1\let\bxuw\pgfplotsretval
  \edef\bxplot{\noexpand\addplot[#3, boxplot prepared={lower whisker=\bxlw,
    lower quartile=\bxlq, median=\bxmed, upper quartile=\bxuq, upper whisker=\bxuw}]
    coordinates {};}%
  \bxplot}
